# Step 1: Import libraries
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.naive_bayes import MultinomialNB
from sklearn.metrics import accuracy_score

# Step 2: Load dataset (create simple dataset if not available)
data = {
    'label': ['spam', 'ham', 'spam', 'ham', 'spam', 'ham'],
    'message': [
        'Win money now',
        'Hello how are you',
        'Claim your prize',
        'Let us meet tomorrow',
        'Free offer just for you',
        'Good morning'
    ]
}

df = pd.DataFrame(data)

# Convert labels to numbers
df['label_num'] = df.label.map({'ham': 0, 'spam': 1})

# Step 3: Split data
X_train, X_test, y_train, y_test = train_test_split(
    df['message'], df['label_num'], test_size=0.2, random_state=42
)

# Step 4: Convert text to vectors
vectorizer = CountVectorizer()
X_train_count = vectorizer.fit_transform(X_train)
X_test_count = vectorizer.transform(X_test)

# Step 5: Train model
model = MultinomialNB()
model.fit(X_train_count, y_train)

# Step 6: Predict
y_pred = model.predict(X_test_count)

# Step 7: Accuracy
print("Accuracy:", accuracy_score(y_test, y_pred))

# Step 8: Test with your own message
msg = ["Congratulations! You won a free ticket"]
msg_count = vectorizer.transform(msg)
prediction = model.predict(msg_count)

if prediction[0] == 1:
    print("Spam Message")
else:
    print("Not Spam")
